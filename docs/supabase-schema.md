### care_guides.care_guide_related_with_fallback

_view_

| Column | Type | Nullable | Default | Notes |
| --- | --- | --- | --- | --- |
| `care_guide_id` | uuid | yes |  |  |
| `related_guide_id` | uuid | yes |  |  |
| `related_slug` | text | yes |  |  |
| `related_name` | text | yes |  |  |
| `sort_order` | integer | yes |  |  |
| `is_manual` | boolean | yes |  |  |

Definition:

```sql
WITH manual_related AS (
         SELECT cgr.care_guide_id,
            related.id AS related_guide_id,
            related.slug AS related_slug,
            ts.species_common_name AS related_name,
            cgr.sort_order,
            true AS is_manual
           FROM care_guides.care_guide_related cgr
             JOIN care_guides.care_guides related ON related.id = cgr.related_guide_id
             JOIN public.turtle_species ts ON ts.id = related.species_id
          WHERE related.status = 'published'::care_guides.care_guide_status
        ), has_manual AS (
         SELECT DISTINCT manual_related.care_guide_id
           FROM manual_related
        ), auto_related AS (
         SELECT cg.id AS care_guide_id,
            other.id AS related_guide_id,
            other.slug AS related_slug,
            ts_other.species_common_name AS related_name,
            0 AS sort_order,
            false AS is_manual
           FROM care_guides.care_guides cg
             JOIN public.turtle_species ts ON ts.id = cg.species_id
             JOIN public.turtle_species ts_other ON ts_other.tax_parent_genus = ts.tax_parent_genus AND ts_other.id <> ts.id
             JOIN care_guides.care_guides other ON other.species_id = ts_other.id AND other.status = 'published'::care_guides.care_guide_status
          WHERE NOT (cg.id IN ( SELECT has_manual.care_guide_id
                   FROM has_manual))
        )
 SELECT manual_related.care_guide_id,
    manual_related.related_guide_id,
    manual_related.related_slug,
    manual_related.related_name,
    manual_related.sort_order,
    manual_related.is_manual
   FROM manual_related
UNION ALL
 SELECT auto_related.care_guide_id,
    auto_related.related_guide_id,
    auto_related.related_slug,
    auto_related.related_name,
    auto_related.sort_order,
    auto_related.is_manual
   FROM auto_related;
```

              from pg_catalog.pg_constraint k
              where k.conrelid = c.oid

#### public.calculate_hot_score(p_score integer, p_created_at timestamp with time zone)

Returns `double precision`

```sql
CREATE OR REPLACE FUNCTION public.calculate_hot_score(p_score integer, p_created_at timestamp with time zone)
 RETURNS double precision
 LANGUAGE plpgsql
 IMMUTABLE
AS $function$
DECLARE
  order_val DOUBLE PRECISION;
  sign_val INT;
  seconds DOUBLE PRECISION;
BEGIN
  -- Determine the sign and order based on score
  IF p_score > 0 THEN
    sign_val := 1;
    order_val := LOG(GREATEST(ABS(p_score), 1));
  ELSIF p_score < 0 THEN
    sign_val := -1;
    order_val := LOG(GREATEST(ABS(p_score), 1));
  ELSE
    sign_val := 0;
    order_val := 0;
  END IF;

  -- Calculate seconds since January 1, 2024 (our epoch)
  -- 1704067200 = Unix timestamp for 2024-01-01 00:00:00 UTC
  seconds := EXTRACT(EPOCH FROM p_created_at) - 1704067200;

  -- Combine score and time factors
  -- 45000 seconds (~12.5 hours) determines how fast new content rises
  RETURN ROUND((sign_val * order_val + seconds / 45000)::NUMERIC, 7);
END;
$function$
```

#### public.get_species_geojson(p_species_id integer, p_status text)

Returns `jsonb`

```sql
CREATE OR REPLACE FUNCTION public.get_species_geojson(p_species_id integer, p_status text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
AS $function$
DECLARE
    result JSONB;
BEGIN
    WITH distribution_features AS (
        SELECT
            sd.origin,
            gr.name AS region_name,
            gr.level AS region_level,
            ts.species_common_name AS species_name,
            ts.species_scientific_name AS scientific_name,
            gr.geometry,
            jsonb_array_elements(gr.geojson->'features') AS feat
        FROM
            species_distributions sd
        JOIN
            geographic_regions gr ON sd.region_id = gr.id
        JOIN
            turtle_species ts ON sd.species_id = ts.id
        WHERE
            sd.species_id = p_species_id
            AND gr.geometry IS NOT NULL
            AND gr.geojson IS NOT NULL
    ),
    combined_geometries AS (
        SELECT
            ST_Collect(geometry) AS all_geom
        FROM
            distribution_features
        WHERE geometry IS NOT NULL
    ),
    bbox_values AS (
        SELECT
            CASE 
                WHEN all_geom IS NOT NULL THEN ST_XMin(all_geom)
                ELSE NULL
            END AS min_lng,
            CASE 
                WHEN all_geom IS NOT NULL THEN ST_YMin(all_geom)
                ELSE NULL
            END AS min_lat,
            CASE 
                WHEN all_geom IS NOT NULL THEN ST_XMax(all_geom)
                ELSE NULL
            END AS max_lng,
            CASE 
                WHEN all_geom IS NOT NULL THEN ST_YMax(all_geom)
                ELSE NULL
            END AS max_lat
        FROM
            combined_geometries
    ),
    combined_features AS (
        SELECT
            jsonb_agg(
                jsonb_build_object(
                    'type', 'Feature',
                    'geometry', feat->'geometry',
                    'properties',
                    jsonb_build_object(
                        'origin', origin,
                        'species_name', species_name,
                        'scientific_name', scientific_name,
                        'region_name', region_name,
                        'region_level', region_level
                    ) || (feat->'properties')
                )
            ) AS all_features
        FROM
            distribution_features
        WHERE feat->'geometry' IS NOT NULL
    )
    SELECT
        jsonb_build_object(
            'type', 'FeatureCollection',
            'bbox', CASE 
                WHEN bv.min_lng IS NOT NULL AND bv.min_lat IS NOT NULL 
                     AND bv.max_lng IS NOT NULL AND bv.max_lat IS NOT NULL
                THEN jsonb_build_array(bv.min_lng, bv.min_lat, bv.max_lng, bv.max_lat)
                ELSE NULL
            END,
            'features', COALESCE(cf.all_features, '[]'::jsonb)
        )
    INTO result
    FROM
        combined_features cf,
        bbox_values bv;

    RETURN result;
END;
$function$
```
